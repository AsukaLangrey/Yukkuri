#!/usr/bin/env python3
"""Assemble c04 from this run's evidence only. Does not run tests or publish.

Final inputs: evidence/test-completion.json (or terminal round-lifecycle),
environment.json, main-at-end.json, native-observation.json and completed runtime
logs. --draft permits absent terminal inputs, writes template_only=true, and does
not write testing/latest/dot.json. Re-run without --draft only after tests stop.

import argparse
import copy
import datetime as dt
import hashlib
import json
import pathlib
import re
import shutil
import zipfile

B = pathlib.Path('/workspace/shared/yukkuri-cloud-20261029T154437-8238c6e')
SRC, RAW = B / 'source', B / 'evidence'
RUN_ID = '20261029T154437Z-8238c6e6380f-c04'
SOURCE = '8238c6e6380fd605ec84cc5267f38de875f6d647'
TREE = 'dc50f13c1d4a9ea2da7c48ae828b942eb612fb7c'
REL = pathlib.Path('testing/results/dot') / RUN_ID
STAGING = B / 'publication/report-staging'
OUT, EV = STAGING / REL, STAGING / REL / 'evidence'
PURE = ['Simulation.Smoke', 'Milestone.Smoke', 'Interaction.Smoke',
        'Input.Intent', 'NaturalLife.Smoke', 'Interaction.Continuity',
        'ActorGeometry.Smoke', 'NaturalLife.Soak']
ALLOWED_STATUSES = {'passed', 'failed', 'blocked', 'skipped', 'not_run', 'timeout', 'interrupted'}

def read(p):
    return json.loads(pathlib.Path(p).read_text())

def save(p, d):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')

def digest(data):
    return hashlib.sha256(data).hexdigest()

def text(p):
    return pathlib.Path(p).read_text().strip()

def optional(name, fallback=None):
    p = RAW / name
    return read(p) if p.exists() else fallback

def blank(check_id, status, reason, evidence=None):
    return {'id': check_id, 'status': status, 'command': None,
            'working_directory': str(SRC), 'started_at': None,
            'finished_at': None, 'exit_code': None, 'passed_count': None,
            'failed_count': None, 'seeds': [], 'reason': reason,
            'evidence': evidence or []}

def shell_step(check_id, stem, command, reason, require_ready=False, import_errors=False):
    names = [stem + '-start.txt', stem + '-end.txt', stem + '-exit.txt', stem + '.txt']
    if not all((RAW / n).is_file() for n in names):
        return blank(check_id, 'not_run', '本轮完整开始/结束/退出码证据尚未齐备。')
    code = int(text(RAW / (stem + '-exit.txt')))
    log = (RAW / (stem + '.txt')).read_text()
    status = 'passed' if code == 0 else 'timeout' if code in {124, 137, 143} else 'failed'
    if require_ready and code == 0 and 'INTERACTION_ROOM_READY' not in log:
        status = 'failed'
        reason += ' 日志没有 INTERACTION_ROOM_READY。'
    if import_errors and re.search(r'(^|\n)ERROR:', log):
        status = 'blocked'
    c = blank(check_id, status, reason, ['evidence/' + n for n in names])
    c.update(command=command, started_at=text(RAW / names[0]),
             finished_at=text(RAW / names[1]), exit_code=code,
             passed_count=1 if status == 'passed' else None,
             failed_count=0 if status == 'passed' else None)
    c['timeout_seconds'] = 300 if stem != 'game-runtime' else 1800
    return c

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--draft', action='store_true')
    args = parser.parse_args()
    run, life = read(RAW / 'run.json'), read(RAW / 'round-lifecycle.json')
    request, report = read(SRC / 'testing/request.json'), read(SRC / 'testing/report-template.json')
    manifest = read(RAW / 'materialization-manifest.json')
    lfs = read(RAW / 'lfs-audit.json')
    assert run['source_commit'] == manifest['source_commit'] == SOURCE
    assert run['source_tree'] == manifest['source_tree'] == TREE
    assert request['revision'] == 'dot-v4-static-polish-20261029'
    assert run['finished_at'] and run['materialized_sources_unchanged']
    completion = optional('test-completion.json') or life
    finished = completion.get('finished_at')
    env = optional('environment.json')
    main_end = optional('main-at-end.json')
    native = optional('native-observation.json')
    missing_inputs = []
    if not finished:
        missing_inputs.append('evidence/test-completion.json: finished_at')
    if completion is not life and completion.get('test_processes_finished') is not True:
        missing_inputs.append('evidence/test-completion.json: test_processes_finished=true')
    if not env:
        missing_inputs.append('evidence/environment.json')
    if not main_end:
        missing_inputs.append('evidence/main-at-end.json')
    if not native:
        missing_inputs.append('evidence/native-observation.json')
    if not args.draft:
        assert not missing_inputs, missing_inputs

    # may change after known asset import failures; product code may not change.

                       'src/Yukkuri.Game/Assets/pipeline-probe.glb.import'}
    changed, source_rows, exact_lfs = [], [], 0
    for f in manifest['files']:
        p = SRC / f['path']
        assert p.is_file(), ('materialized file missing', f['path'])
        data = p.read_bytes()
        sha = digest(data)
        blob = hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()
        assert f['git_blob_sha'] == expected[f['path']], ('manifest tree mismatch', f['path'])
        if sha != f['sha256']:
            changed.append(f['path'])
        if f.get('lfs_restored'):
            assert len(data) == f['lfs_size'] and sha == f['lfs_oid'], ('LFS mismatch', f['path'])
            exact_lfs += 1
        elif f['path'] not in generated_allow:
            assert blob == expected[f['path']], ('source mismatch', f['path'])
        source_rows.append({'path': f['path'], 'expected_git_blob_sha': expected[f['path']],
                            'actual_git_blob_sha': blob, 'sha256': sha,
                            'lfs_restored': f.get('lfs_restored', False)})
    assert set(changed) <= generated_allow, ('Unexpected product/source changes', changed)
    assert exact_lfs == 12 and len(lfs['missing_original_objects']) == 3

    steps = {s['id']: s for s in run['steps']}
    pure_checks = []
    failures = []
    for name in PURE:
        c = copy.deepcopy(steps[name])
        assert c['finished_at'] and c['exit_code'] is not None
        p = RAW / (name + '.json')
        log = (RAW / (name + '.txt')).read_text()
        if p.exists():
            d = read(p)
            rows = next(d[k] for k in ['checks', 'results', 'scenarios'] if isinstance(d.get(k), list))
            passed = sum(r.get('passed') is True for r in rows)
            failed = sum(r.get('passed') is False for r in rows)

                assert d['failures'] == failed
            failures += [{'check_id': name, **r} for r in rows if r.get('passed') is False]
        else:
            passed = len(re.findall(r'^PASS[: ]', log, re.M))
            failed = len(re.findall(r'^FAIL[: ]', log, re.M))
        assert (passed, failed) == (c['passed_count'], c['failed_count']), name
        assert c['status'] == ('failed' if failed or c['exit_code'] else 'passed'), name
        c['restore'] = copy.deepcopy(steps[name + '-restore'])
        c['evidence'] += c['restore']['evidence']
        c['timeout_seconds'] = 1800 if name == 'NaturalLife.Soak' else 600
        c['reason'] = '本轮独立构建并实际执行；退出码及 JSON/日志重新核对。仅纯 .NET 规则/数值检查，不等于原生 UI 或玩法验收。'
        c['seeds'] = {
            'Simulation.Smoke': [92137, 20261028],
            'Milestone.Smoke': [92137, 20261027] + list(range(1, 121)),
            'Interaction.Smoke': [20261028], 'Input.Intent': [20261028],
            'NaturalLife.Smoke': [20261028, 20261029, 1009, 779, 292107, 291044] + list(range(1, 21)),
            'Interaction.Continuity': [20261028] + list(range(71001, 71013)),
            'ActorGeometry.Smoke': [20261028, 20261029],
            'NaturalLife.Soak': [72341, 592813, 1597331, 20261029]
        }[name]
        c['seed_provenance'] = 'evidence/seed-provenance.json；包含默认世界种子、显式固定种子和固定种子生成器。'
        pure_checks.append(c)

    failed = sum(c['failed_count'] for c in pure_checks)

    assert len(failures) == 1 and failures[0]['check_id'] == 'Interaction.Continuity'

        if f.name == 'NaturalLife.Soak.json':

                'omitted_fields': ['successful scenarios[].snapshots'],
                'raw_sha256': digest(f.read_bytes()), 'raw_bytes': f.stat().st_size,

            save(dest, slim)
        else:

                 'testing/report-template.json', 'testing/DOT-HANDOFF.md',

        'scope': 'Every materialized file independently rechecked against pinned tree/blob manifest; recovered LFS bytes checked against original pointer OID and size. Generated import metadata changes remain only in test copy.'})
    seed_patterns = re.compile(r'CreateInteractionPlaytest|CreateMilestone|RandomState|new Random|for\(.*seed|new ulong|ulong seed|Seed\s*=')
    seed_lines = {}
    for name in PURE:
        p = SRC / 'checks' / name / 'Program.cs'

    for name in ['src/Yukkuri.Simulation/InteractionWorld.cs', 'src/Yukkuri.Simulation/MilestoneWorld.cs',
                 'src/Yukkuri.Simulation/WorldSimulation.cs', 'src/Yukkuri.Simulation/SimulationConfig.cs',
                 'src/Yukkuri.Simulation/WorldState.cs']:

        if p.exists():
            seed_lines[name] = [{'line': i, 'text': line} for i, line in enumerate(p.read_text().splitlines(), 1) if seed_patterns.search(line)]

         'interpretation': 'Explicit seed-generator values enumerate reproducible sampled cases, not every generated random value. Soak uses four fixed seeds and six scenarios.'})
    save(EV / 'failure-extract.json', {'source_commit': SOURCE, 'failures': failures,
         'no_failure_retry': True, 'root_cause_confirmed': False,
         'compound_assertion': 'accepted && sawSwallow && moved && c.X > .5',
         'source_location': 'checks/Interaction.Continuity/Program.cs:143',
         'subcase_execution': {
             'counting_unit': 'One named top-level check; nested combinations are not added to the 502/1 aggregate.',
             'control_flow_inference': 'The seven preceding combinations reached their end before Drink/water. Drink/water reached the terminal compound assertion, then threw. Drink/rest was not entered.',
             'not_executed_after_failure': ['Drink/water natural-intake completion assertion at line 144',
                 'Drink/water cancellation and explicit put-down assertions at lines 145-148', 'Drink/rest']},
         'interpretation_limit': 'Current JSON only records the compound assertion message, not component values. A terminal X condition after 240 * 0.05 s cannot alone distinguish no movement from moving then returning. Product-trapping root cause is unconfirmed.'})

    dotnet = '/workspace/shared/yukkuri-cloud-20261029/toolchain/dotnet/dotnet'
    restore = shell_step('game-restore', 'game-restore',
        f'{dotnet} restore src/Yukkuri.Game/Yukkuri.Game.csproj --configfile {B}/NuGet.godot.Config -p:NuGetAudit=false',
        '本轮官方包隔离缓存完成 restore，准确 SDK/包身份见证据。')
    build = shell_step('game-build', 'game-build',
        f'{dotnet} build src/Yukkuri.Game/Yukkuri.Game.csproj --configuration Debug --no-restore -p:UseSharedCompilation=false',
        '本轮固定源码 Linux Debug 游戏构建；0 warning / 0 error。不是 Windows 打包验证。')
    build['restore'] = restore
    build['evidence'] += restore['evidence'] + ['evidence/godot-package-identity.json', 'evidence/nuget-package-identities.json']
    imp = shell_step('asset-import', 'game-import',
        '$GODOT_BIN --headless --path "$GAME_PATH" --editor --import --quit',
        '导入进程退出 0 但标题 PNG / probe GLB 指针造成明确 ERROR；完整素材导入 blocked，不能计全通过。', import_errors=True)
    runtime = shell_step('runtime-smoke', 'game-runtime',
        f'bash {B}/run-game.sh (launched through authorized cloud CUA terminal)',
        '只证明本轮固定源码游戏进入 InteractionRoom 并结束；实际交互范围另见 native-observation。', require_ready=True)
    runtime.update(working_directory=None,
        working_directory_note='CUA terminal cwd was not recorded; env.sh and run-game.sh did not cd. Absolute GAME_PATH supplied by --path.',
        project_path=str(SRC / 'src/Yukkuri.Game'))
    runtime['evidence'] += ['evidence/native-observation.json'] if native else []
    lfs_check = blank('lfs-integrity', 'blocked',
        '本轮 15 个唯一原始对象中 12 个按原 LFS SHA256/字节数恢复并重核；标题 PNG、probe GLB、源 .blend 仍缺。完整仓库 LFS 未通过。',
        ['evidence/lfs-audit.json', 'evidence/materialization-manifest.json', 'evidence/source-verification.json'])
    lfs_check.update(command=f'python {B}/report-preparation/assemble_report.py (source and LFS audit during report assembly; materialization audit retained separately)',
                     started_at=lfs.get('checked_at'), finished_at=lfs.get('checked_at'),
                     exit_code=0, recovered_count=12, unique_objects_checked=15,
                     exit_code_scope='审计完成；不是完整 LFS 通过。')
    save_check = blank('Save.Compatibility', 'blocked',
        '没有获准历史存档样本。新建隔离存档、合成缺字段/损坏样本均不替代真实历史兼容。')
    visual = blank('visual-interaction-review', 'not_run',
        '未完成全部 v4 与既有视觉焦点。有限真实观察单列，不能由纯逻辑、启动或截图概括为完整验收。',
        ['evidence/native-observation.json'] if native else [])
    if native and native.get('full_review_passed') is True:
        visual.update(status='passed', reason=native.get('full_review_reason', '观察记录明确确认全部要求已覆盖。'))
    elif native and native.get('full_review_reason'):
        visual['reason'] = native['full_review_reason']
    mapping = {c['id']: c for c in [lfs_check, build, *pure_checks, save_check, runtime, visual]}
    checks = [mapping[k] for k in request['required_checks']]
    checks.append(imp)
    if native:
        nstatus = native.get('status', 'not_run')
        assert nstatus in ALLOWED_STATUSES
        supplemental = blank('native-observed-interaction', nstatus,
            native.get('summary', native.get('reason', '仅记录中的实际操作和观察成立；未观察部分不计通过。')),
            ['evidence/native-observation.json'])
        supplemental.update(command=native.get('command', 'CUA/Sky native mouse and keyboard; exact sequence in native-observation.json'),
                            started_at=native.get('started_at'), finished_at=native.get('finished_at'),
                            passed_count=native.get('passed_count'), failed_count=native.get('failed_count'),
                            required=False)
        supplemental.update(working_directory=None,
            working_directory_note='CUA terminal cwd was not recorded; env.sh and run-game.sh did not cd. Absolute GAME_PATH supplied by --path.',
            project_path=str(SRC / 'src/Yukkuri.Game'))
        checks.append(supplemental)
    additional = optional('additional-checks.json', [])
    if isinstance(additional, dict):
        additional = additional.get('checks', [])
    for c in additional:
        assert c['status'] in ALLOWED_STATUSES and c['id'] not in {x['id'] for x in checks}
        assert all(k in c for k in ['command', 'started_at', 'finished_at', 'exit_code', 'reason', 'evidence'])
        c = copy.deepcopy(c)
        c.setdefault('working_directory', None)
        c.setdefault('working_directory_note', 'CUA terminal cwd was not recorded; wrapper did not cd. Absolute project path supplied by --path.')
        c.setdefault('project_path', str(SRC / 'src/Yukkuri.Game'))
        c.setdefault('seeds', [])
        c.setdefault('timeout_seconds', 300)
        if c['id'] == 'engine-contact':
            d = read(RAW / 'contact.json')
            assert int(text(RAW / 'engine-contact-exit.txt')) == c['exit_code'] == 0
            assert d['failures'] == 0 and len(d['checks']) == 4
            assert all(x['passed'] is True for x in d['checks'])
            assert c['passed_count'] == 4 and c['failed_count'] == 0
            assert text(RAW / 'engine-contact-start.txt') == c['started_at']
            assert text(RAW / 'engine-contact-end.txt') == c['finished_at']
            c['evidence'] += ['evidence/engine-contact-start.txt', 'evidence/engine-contact-end.txt', 'evidence/engine-contact-exit.txt']
        elif c['id'] == 'engine-layout':
            d = read(RAW / 'render-checks.json')
            assert int(text(RAW / 'engine-render-exit.txt')) == c['exit_code'] == 0
            assert len(d['results']) == 5
            assert all(x['bottomFits'] is True and x['forceSingleLine'] is True for x in d['results'])
            assert c['passed_count'] == 10 and c['failed_count'] == 0
            assert text(RAW / 'engine-render-start.txt') == c['started_at']
            assert text(RAW / 'engine-render-end.txt') == c['finished_at']
            c['performance'] = d.get('performance')
            c['evidence'] += ['evidence/engine-render-start.txt', 'evidence/engine-render-end.txt',
                'evidence/engine-render-exit.txt', 'evidence/render-visual-review.json',
                'evidence/image-manifest.json', 'evidence/render-library.json']
        checks.append(c)
    audio = blank('audio-playback', 'blocked',
        '没有本轮有效可听音频验收；Dummy/未监听不能证明音效或声画反馈。',
        ['evidence/game-runtime.txt'] if (RAW / 'game-runtime.txt').exists() else [])
    if not any(c['id'] == 'audio-playback' for c in checks):
        checks.append(audio)

    current_main = None
    if main_end:
        current_main = main_end.get('main_commit') or main_end.get('commit') or main_end.get('sha')
        if not current_main and isinstance(main_end.get('main'), dict):
            current_main = main_end['main'].get('commit') or main_end['main'].get('sha')
        if not current_main:
            current_main = main_end.get('source_commit')
    if not args.draft:
        assert current_main and re.fullmatch('[0-9a-f]{40}', current_main), ('unrecognized main-at-end schema', main_end)
        assert runtime['status'] != 'not_run', 'Runtime is missing its terminal artifacts.'
    summary = f'8 套纯 .NET 检查共 {passed} 通过、{failed} 失败；连续性新增 Drink/water 复合断言失败，根因未确认。游戏构建通过；真实运行和有限观察按本轮证据单列。完整素材、历史样本、声音及未覆盖 UI 仍有缺项，总状态 failed，不是整体验收。'
    report.update(template_only=args.draft, run_id=RUN_ID, source_commit=SOURCE, source_tree=TREE,
        request_revision=request['revision'], request_sha256=digest((SRC / 'testing/request.json').read_bytes()),
        protocol_sha256=digest((SRC / 'testing/PROTOCOL.md').read_bytes()),
        template_sha256=digest((SRC / 'testing/report-template.json').read_bytes()),
        agents_sha256=digest((SRC / 'AGENTS.md').read_bytes()), source_worktree_clean=None,
        started_at=life['started_at'], finished_at=finished, retry_of=None, supersedes=None,
        environment=env or {'id': 'draft-awaiting-current-environment-evidence'}, status='failed',
        summary=summary, checks=checks, findings=[
            {'id': 'continuity-held-drink-water', 'severity': 'test-failure', 'status': 'confirmed',
             'title': '持续持有已完成饮水工具的 Drink/water 检查失败',
             'details': failures[0], 'root_cause_confirmed': False,
             'interpretation': '合取条件 accepted&&sawSwallow&&moved&&c.X>.5 在12秒末失败；未打印各分量。静态分析存在补水完成后返回旧目标而导致终点 X 条件失败的可能，不能由当前记录确认产品被困住。',
             'coverage_limit': '这是1个具名检查失败，不是9个子组合全部失败。异常抛出后Drink/water的后续自然摄取/取消/放下断言未执行；Drink/rest未进入。前7子组合完成仅由固定循环顺序与异常控制流推得，不增加顶层计数。',
             'evidence': ['evidence/Interaction.Continuity.txt', 'evidence/Interaction.Continuity.json', 'evidence/failure-extract.json']},
