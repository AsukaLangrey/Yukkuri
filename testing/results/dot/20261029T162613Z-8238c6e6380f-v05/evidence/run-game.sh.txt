#!/usr/bin/env bash
set -u
source /workspace/shared/yukkuri-cloud-20261029T154437-8238c6e/env.sh

export XDG_DATA_HOME="$B/data" XDG_CONFIG_HOME="$B/config" XDG_CACHE_HOME="$B/cache" XDG_RUNTIME_DIR="$B/runtime" HOME="$B/home" TMPDIR="$B/temp"

exit "$s"
