#!/usr/bin/env bash
set -u
B=/workspace/shared/yukkuri-cloud-20261030T102053-2da4423
source "$B/env.sh"
mkdir -p "$B/render"
pwd > "$B/evidence/engine-desktop-cwd.txt"
for test in contact render; do
 date -u '+%Y-%m-%dT%H:%M:%SZ' > "$B/evidence/engine-$test-start.txt"
 if [ "$test" = contact ]; then args=(--contact-check-output "$B/evidence/contact.json"); else args=(--room-render-output "$B/render"); fi
 timeout --signal=TERM --kill-after=5 300 "$GODOT_BIN" --path "$GAME_PATH" --audio-driver Dummy --resolution 1280x800 -- --interaction-test-profile "cloud1030-c10-$test" "${args[@]}" > "$B/evidence/engine-$test.txt" 2>&1
 s=$?
 printf '%s\n' "$s" > "$B/evidence/engine-$test-exit.txt"
 date -u '+%Y-%m-%dT%H:%M:%SZ' > "$B/evidence/engine-$test-end.txt"
done
