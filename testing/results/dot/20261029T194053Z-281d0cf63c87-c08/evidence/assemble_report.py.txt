def read(p):
    return json.loads(pathlib.Path(p).read_text())

def save(p, d):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')

def digest(data):
    return hashlib.sha256(data).hexdigest()

def text(p):
    return pathlib.Path(p).read_text().strip()

def optional(name, fallback=None):
    p = RAW / name
    return read(p) if p.exists() else fallback

def blank(check_id, status, reason, evidence=None):
    return {'id': check_id, 'status': status, 'command': None,
            'working_directory': None, 'started_at': None,
            'finished_at': None, 'exit_code': None, 'passed_count': None,
            'failed_count': None, 'seeds': [], 'reason': reason,
            'evidence': evidence or []}

def shell_step(check_id, stem, command, reason, require_ready=False, import_errors=False):
    names = [stem + '-start.txt', stem + '-end.txt', stem + '-exit.txt', stem + '.txt']
    if not all((RAW / n).is_file() for n in names):
        return blank(check_id, 'not_run', '本轮完整开始/结束/退出码证据尚未齐备。')
    code = int(text(RAW / (stem + '-exit.txt')))
    log = (RAW / (stem + '.txt')).read_text()
    status = 'passed' if code == 0 else 'timeout' if code in {124, 137, 143} else 'failed'
    if require_ready and code == 0 and 'INTERACTION_ROOM_READY' not in log:
        status = 'failed'
        reason += ' 日志没有 INTERACTION_ROOM_READY。'
    if import_errors and re.search(r'(^|\n)ERROR:', log):
        status = 'blocked'
    c = blank(check_id, status, reason, ['evidence/' + n for n in names])
    c.update(command=command, working_directory=str(SRC), started_at=text(RAW / names[0]),
             finished_at=text(RAW / names[1]), exit_code=code,
             passed_count=1 if status == 'passed' else None,
             failed_count=0 if status == 'passed' else None)
    c['timeout_seconds'] = 300 if stem != 'game-runtime' else 1800
    return c
